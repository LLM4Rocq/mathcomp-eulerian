\chapter{Cycles, inversions, and the major index}
\label{ch:inversions}

This chapter formalizes Stanley EC1 \S1.3 (cycles and inversions), the
section that introduces the two foundational permutation statistics
beyond descents: the \emph{cycle count} $c(w)$ and the two Mahonian
statistics \emph{inv} and \emph{maj}.  It then covers Foata's first
fundamental bijection and MacMahon's \emph{inv-maj equidistribution},
the $q$-factorial generating function and the bivariate $q$-Eulerian
polynomial from \S1.4, and the longest alternating subsequence
characterization from \S1.6.2.  A final section records the partial
formalization of \S1.6.4 (André's reflection method): the headline
recurrence
$2 E_{n+2} = \sum_k \binom{n+1}{k} E_k E_{n+1-k}$ is proven modulo
one named admit, in the out-of-build-chain file
\code{experimental/reflection.v}.

\medskip
This chapter uses the same indexing convention as
Chapter~\ref{ch:descents}: Stanley's $w \in \mathfrak{S}_n$ becomes our
$\sigma \in \Snp{n+1}$, and Stanley's descent positions $1, \dots, n-1$
become our $\{0, \dots, n-1\}$ via the index shift.

% ============================================================
\section{Cycle representation of permutations}
% ============================================================

\begin{definition}[Cycle count]
  \label{def:cycle_count}
  \rocq{mathcomp_eulerian.cycles.cycle_count}
  \rocqok
  For a permutation $\sigma \in \Snp{n}$, the \emph{cycle count}
  $c(\sigma)$ is the number of cycles in the disjoint-cycle
  decomposition of $\sigma$.  In MathComp this is the cardinality of
  $\mathrm{porbits}(\sigma)$, the set of orbits of $\sigma$ acting on
  $\{0, \dots, n-1\}$ by iteration.
\end{definition}

\begin{lemma}[$c(\sigma) \leq n$]
  \label{lem:cycle_count_le}
  \uses{def:cycle_count}
  \rocq{mathcomp_eulerian.cycles.cycle_count_le}
  \rocqok
  Every cycle is non-empty, so the cycle count is bounded by $n$.
\end{lemma}

\begin{lemma}[Identity has $n$ cycles]
  \label{lem:cycle_count_id}
  \uses{def:cycle_count}
  \rocq{mathcomp_eulerian.cycles.cycle_count_id}
  \rocqok
  Each fixed point is a 1-cycle, so $c(\mathrm{id}) = n$.
\end{lemma}

\section{Stirling numbers of the first kind}
% ============================================================

\begin{definition}[Signless Stirling cycle numbers]
  \label{def:stirling_c}
  \uses{def:cycle_count}
  \rocq{mathcomp_eulerian.cycles.stirling_c}
  \rocqok
  $c(n, k) = \#\{\sigma \in \Snp{n} : c(\sigma) = k\}$ counts
  permutations of $\{0, \dots, n-1\}$ with exactly $k$ cycles in their
  disjoint-cycle decomposition.  Stanley writes $c(n, k)$ throughout
  \S1.3.2.
\end{definition}

\begin{lemma}[Row sum]
  \label{lem:stirling_c_row_sum_fact}
  \uses{def:stirling_c}
  \rocq{mathcomp_eulerian.cycles.stirling_c_row_sum_fact}
  \rocqok
  $\sum_{k=0}^{n} c(n, k) = n!$.  Each permutation is counted by
  exactly one $k$, so the row sum recovers the total.
\end{lemma}

\begin{theorem}[Stirling recurrence]
  \label{thm:stirling_c_rec}
  \uses{def:stirling_c}
  \rocq{mathcomp_eulerian.stirling_fiber.stirling_c_rec}
  \rocqok
  For all $n, k \in \NN$,
  \[
    c(n+1, k+1) \;=\; n \cdot c(n, k+1) + c(n, k).
  \]
  Combinatorially: $n+1$ either becomes a fixed point of a
  $k$-cycle permutation (giving $c(n, k)$), or is inserted into one
  of $n$ positions of a $(k+1)$-cycle permutation (giving
  $n \cdot c(n, k+1)$).

  \emph{Formalization route.}  The proof factors through a
  construction $\mathrm{insert\_after}\,j_0\,s : \Snp{n+1}$ that
  splices $\mathrm{ord\_max}$ into the $s$-cycle through $j_0$.
  The key algebraic identity
  \[
    \mathrm{insert\_after}\,j_0\,s
    \;=\;
    \mathrm{tperm}\,(\mathrm{ord\_max})\,(\mathrm{lift}\,\mathrm{ord\_max}\,j_0)
    \;\cdot\;
    \mathrm{lift\_perm}\,\mathrm{ord\_max}\,\mathrm{ord\_max}\,s
  \]
  reduces cycle-count preservation to MathComp's
  \texttt{porbits\_mul\_tperm} composed with the fixed-point
  cycle-count lemma in
  \rocq{mathcomp_eulerian.cycles_rec.cycle_count_lift_perm_id}.
\end{theorem}

\section{Inversions}
% ============================================================

\begin{definition}[Inversion at a pair]
  \label{def:is_inv}
  \rocq{mathcomp_eulerian.inversions.is_inv}
  \rocqok
  Position pair $(i, j)$ is an \emph{inversion} of $\sigma$ when
  $i < j$ but $\sigma(j) < \sigma(i)$ — i.e., the pair appears in the
  ``wrong'' order.
\end{definition}

\begin{definition}[Inversion set and count]
  \label{def:inv}
  \uses{def:is_inv}
  \rocq{mathcomp_eulerian.inversions.inv_set}
  \rocq{mathcomp_eulerian.inversions.inv}
  \rocqok
  $\mathrm{inv\_set}(\sigma)$ is the set of inversion pairs;
  $\mathrm{inv}(\sigma) = |\mathrm{inv\_set}(\sigma)|$ is their count.
  Stanley writes $\mathrm{inv}(w)$.
\end{definition}

\begin{lemma}[Identity has no inversions]
  \label{lem:inv_id}
  \uses{def:inv}
  \rocq{mathcomp_eulerian.inversions.inv_id}
  \rocqok
  $\mathrm{inv}(\mathrm{id}) = 0$.
\end{lemma}

\begin{lemma}[Triangular counting]
  \label{lem:card_ord_pair}
  \rocq{mathcomp_eulerian.inversions.card_ord_pair}
  \rocqok
  $\#\{(i, j) \in \{0, \dots, m-1\}^2 : i < j\} = \binom{m}{2}$.
\end{lemma}

\begin{lemma}[Inversion bound]
  \label{lem:inv_le}
  \uses{def:inv, lem:card_ord_pair}
  \rocq{mathcomp_eulerian.inversions.inv_le}
  \rocqok
  $\mathrm{inv}(\sigma) \leq \binom{n+1}{2}$.  The maximum is achieved by
  the order-reversing permutation, which has every pair inverted.
\end{lemma}

\begin{theorem}[Inversion reversal symmetry]
  \label{thm:inv_rev_perm}
  \uses{def:inv, lem:card_ord_pair, lem:des_rev_perm}
  \rocq{mathcomp_eulerian.inversions.inv_rev_perm}
  \rocqok
  Let $\rev_n \in \Snp{n+1}$ be the order-reversing permutation.  Then
  \[
    \mathrm{inv}(\rev_n \cdot \sigma) \;=\; \binom{n+1}{2} - \mathrm{inv}(\sigma).
  \]
  Each pair $(i, j)$ with $i < j$ is either an inversion or a
  co-inversion of $\sigma$ (exactly one), and the bijection
  $(i, j) \mapsto (\rev_n(j), \rev_n(i))$ swaps the two.
\end{theorem}

\section{The major index}
% ============================================================

\begin{definition}[Major index]
  \label{def:maj}
  \uses{def:descent_set}
  \rocq{mathcomp_eulerian.inversions.maj}
  \rocqok
  The \emph{major index} is the sum of (1-indexed) descent positions:
  \[
    \mathrm{maj}(\sigma) \;=\; \sum_{i \in \DescSet{\sigma}} (i + 1)
    \;=\; \sum_{i \in D(w)} i \quad (\text{Stanley's notation, p.~22}).
  \]
\end{definition}

\begin{lemma}[Identity has zero major index]
  \label{lem:maj_id}
  \uses{def:maj}
  \rocq{mathcomp_eulerian.inversions.maj_id}
  \rocqok
  $\mathrm{maj}(\mathrm{id}) = 0$.
\end{lemma}

\begin{lemma}[Major index bound]
  \label{lem:maj_le}
  \uses{def:maj}
  \rocq{mathcomp_eulerian.inversions.maj_le}
  \rocqok
  $\mathrm{maj}(\sigma) \leq \binom{n+1}{2}$.  The maximum is achieved
  by the order-reversing permutation, where every position is a descent
  and the sum is $1 + 2 + \dots + n = \binom{n+1}{2}$.
\end{lemma}

\begin{theorem}[Major index reversal identity]
  \label{thm:maj_rev_perm}
  \uses{def:maj, def:des_asc, lem:des_rev_perm}
  \rocq{mathcomp_eulerian.inversions.maj_rev_perm}
  \rocqok
  For $\rev_n \in \Snp{n+1}$,
  \[
    \mathrm{maj}(\rev_n \cdot \sigma) + (n + 1) \cdot \des(\sigma)
    \;=\; \binom{n+1}{2} + \mathrm{maj}(\sigma).
  \]
  The naive identity $\mathrm{maj}(\rev_n \cdot \sigma) = \binom{n+1}{2}
  - \mathrm{maj}(\sigma)$ is \emph{false} in general: the asymmetry
  comes from $\mathrm{maj}$'s 1-indexed positions.  The correct
  identity above accounts for it via the per-descent offset
  $(n - i) + (i + 1) = n + 1$.
\end{theorem}

\section{Foata's first fundamental bijection and MacMahon's equidistribution}
% ============================================================

\begin{definition}[Foata's first fundamental bijection]
  \label{def:foata_perm}
  \rocq{mathcomp_eulerian.foata.foata_perm}
  \rocqok
  Foata's first fundamental bijection
  $\varphi : \Snp{n+1} \to \Snp{n+1}$ is defined recursively on
  the seq representation: read $w = w_1 w_2 \dots w_{n+1}$ left-to-right
  and at each step splice the new letter into a current word via a
  cyclic-rotation rule on maximal blocks.  Implemented as
  \texttt{seq\_to\_perm $\circ$ foata $\circ$ perm\_to\_seq}.
\end{definition}

\begin{theorem}[Foata's bijection sends $\mathrm{maj}$ to $\mathrm{inv}$]
  \label{thm:foata_perm_inv_maj}
  \uses{def:foata_perm, def:inv, def:maj}
  \rocq{mathcomp_eulerian.foata.foata_perm_inv_maj}
  \rocqok
  For every $\sigma \in \Snp{n+1}$,
  \[
    \mathrm{inv}(\varphi(\sigma)) \;=\; \mathrm{maj}(\sigma).
  \]
  This is the load-bearing identity of the bijection.  It is proved
  via the seq-level invariant
  \texttt{foata\_inv\_eq\_maj}: $\mathrm{inv\_seq}(\mathrm{foata}(w)) =
  \mathrm{maj\_seq}(w)$, which in turn is proved by reverse induction
  on $w$ using the per-step invariant
  \texttt{foata\_step\_inv} (a tally of cyclic-rotation effects on the
  inversion count).
\end{theorem}

\begin{theorem}[Foata's bijection is bijective]
  \label{thm:foata_perm_inj}
  \uses{def:foata_perm}
  \rocq{mathcomp_eulerian.foata.foata_perm_inj}
  \rocqok
  $\varphi$ is injective (hence bijective on the finite group $\Snp{n+1}$).
  Proof: explicit construction of the inverse $\varphi^{-1}$ on seqs
  (cuts at $P$-elements rather than after them, undoing the cyclic
  rotation block-wise) plus a cancellation lemma
  $\varphi^{-1}(\varphi(w)) = w$.
\end{theorem}

\begin{theorem}[MacMahon's equidistribution of $\mathrm{inv}$ and $\mathrm{maj}$]
  \label{thm:inv_maj_equidistr}
  \uses{def:inv, def:maj, thm:foata_perm_inv_maj, thm:foata_perm_inj}
  \rocq{mathcomp_eulerian.foata.inv_maj_equidistr}
  \rocqok
  \emph{Stanley EC1 Theorem 1.3.4 (MacMahon).}  For every $n, k \in \NN$,
  \[
    \#\{\sigma \in \Snp{n+1} : \mathrm{inv}(\sigma) = k\}
    \;=\;
    \#\{\sigma \in \Snp{n+1} : \mathrm{maj}(\sigma) = k\}.
  \]
  The proof is a direct consequence of \cref{thm:foata_perm_inv_maj}
  and \cref{thm:foata_perm_inj}: $\varphi$ is a bijection of
  $\Snp{n+1}$ that maps $\{\sigma : \mathrm{maj}(\sigma) = k\}$ onto
  $\{\sigma : \mathrm{inv}(\sigma) = k\}$ for every $k$.  The closed
  form for both distributions is the $q$-factorial
  (\cref{thm:inv_q_fact} and \cref{thm:maj_q_fact}).
\end{theorem}

\section{The $q$-factorial generating function}
% ============================================================

\begin{definition}[$q$-integer and $q$-factorial]
  \label{def:q_fact}
  \rocq{mathcomp_eulerian.qfact.q_int}
  \rocq{mathcomp_eulerian.qfact.q_fact}
  \rocqok
  In the polynomial ring $\ZZ[q]$:
  \[
    [n]_q := 1 + q + q^2 + \dots + q^{n-1},
    \qquad
    [n+1]_q! := [1]_q \cdot [2]_q \cdot \dots \cdot [n+1]_q.
  \]
\end{definition}

\begin{theorem}[Inversion generating function]
  \label{thm:inv_q_fact}
  \uses{def:inv, def:q_fact}
  \rocq{mathcomp_eulerian.qfact.inv_q_fact}
  \rocqok
  \emph{Stanley EC1 §1.3.4 / Cor 1.3.10.}  For every $n \in \NN$,
  \[
    \sum_{\sigma \in \Snp{n+1}} q^{\mathrm{inv}(\sigma)} \;=\; [n+1]_q!.
  \]
  Proved by induction on $n$ using the value-based ``insert
  $\mathrm{ord}_{\max}$'' bijection $\Snp{n+2} \simeq \Snp{n+1}
  \times \{0, \dots, n+1\}$ from $\code{eulerian.v}$.  The key step is
  $\mathrm{inv}(\mathrm{insert\_max\_perm}(\tau, p)) = \mathrm{inv}(\tau)
  + (n+1 - p)$.
\end{theorem}

\begin{theorem}[Major-index generating function]
  \label{thm:maj_q_fact}
  \uses{def:maj, def:q_fact, thm:inv_q_fact, thm:foata_perm_inj, thm:foata_perm_inv_maj}
  \rocq{mathcomp_eulerian.qfact.maj_q_fact}
  \rocqok
  \emph{Stanley EC1 §1.3.4.}  For every $n \in \NN$,
  \[
    \sum_{\sigma \in \Snp{n+1}} q^{\mathrm{maj}(\sigma)} \;=\; [n+1]_q!.
  \]
  Immediate corollary of \cref{thm:inv_q_fact} via Foata's bijection.
\end{theorem}

\section{The $q$-Eulerian polynomial — joint $(\mathrm{maj},\des)$ distribution}
% ============================================================

\begin{definition}[$q$-Eulerian polynomial]
  \label{def:q_eul_pol}
  \uses{def:maj, def:des_asc}
  \rocq{mathcomp_eulerian.qeul.q_eul_pol}
  \rocqok
  \emph{Stanley EC1 §1.4.}  The bivariate generating function for the
  joint distribution of $(\mathrm{maj}, \des)$ over $\Snp{n+1}$:
  \[
    A_n(q, t) \;:=\; \sum_{\sigma \in \Snp{n+1}}
                     q^{\mathrm{maj}(\sigma)} \cdot t^{\des(\sigma)}.
  \]
  Formalized in $\ZZ[q][t]$ (mathcomp's \texttt{\{poly \{poly int\}\}}).
\end{definition}

\begin{theorem}[Specialization $t = 1$ — the $q$-factorial]
  \label{thm:q_eul_pol_t1}
  \uses{def:q_eul_pol, def:q_fact, thm:maj_q_fact}
  \rocq{mathcomp_eulerian.qeul.q_eul_pol_t1}
  \rocqok
  $A_n(q, 1) = [n+1]_q!$.
  Setting $t = 1$ collapses the bivariate to the marginal
  generating function for $\mathrm{maj}$, which is the q-factorial
  by \cref{thm:maj_q_fact}.
\end{theorem}

\begin{theorem}[Specialization $q = 1$ — the classical Eulerian polynomial]
  \label{thm:q_eul_pol_q1}
  \uses{def:q_eul_pol, def:eulerian, thm:beta_eulerian}
  \rocq{mathcomp_eulerian.qeul.q_eul_pol_q1}
  \rocqok
  $A_n(1, t) = \mathrm{eul\_pol}_n(t) := \sum_{k=0}^{n} A_{n+1, k+1} \cdot t^k$.
  Setting $q = 1$ drops the $\mathrm{maj}$ weight, leaving the
  classical Eulerian polynomial $\sum_\sigma t^{\des(\sigma)}$.
  Formalized via the rmorphism wrapper $\mathrm{horner\_eval}(1)$
  applied coefficient-wise across the outer polynomial.
\end{theorem}

\section{Longest alternating subsequences (Stanley \S1.6.2)}
% ============================================================

\begin{definition}[Turning-point set]
  \label{def:turn_count}
  \uses{def:descent_set}
  \rocq{mathcomp_eulerian.altsub.is_turn}
  \rocq{mathcomp_eulerian.altsub.turn_count}
  \rocqok
  Position $i \in \{0, \dots, n-1\}$ is a \emph{turning point} of
  $\sigma \in \Snp{n+2}$ if the direction of $\sigma$ changes between
  slot $i$ and slot $i+1$:
  \[
    \mathrm{is\_turn}(\sigma, i)
    \;:=\;
    \mathrm{is\_descent}(\sigma, i) \,\oplus\, \mathrm{is\_descent}(\sigma, i+1).
  \]
  $\mathrm{turn\_count}(\sigma) = \#\{i \in \{0, \dots, n-1\} : \mathrm{is\_turn}(\sigma, i)\}$.
\end{definition}

\begin{definition}[Longest alternating subsequence]
  \label{def:as_perm_max}
  \rocq{mathcomp_eulerian.altsub.as_perm_max}
  \rocqok
  $\mathrm{as}(\sigma)$ is the length of the longest alternating
  subsequence of $\sigma$, formalized as
  \[
    \mathrm{as\_perm\_max}(\sigma)
    \;:=\;
    \max_{I \subseteq \{0, \dots, n+1\}} \{|I| : \text{the subseq of }\sigma\text{ at positions }I\text{ is alternating}\}.
  \]
\end{definition}

\begin{theorem}[Existence direction]
  \label{thm:as_perm_max_lower}
  \uses{def:turn_count, def:as_perm_max}
  \rocq{mathcomp_eulerian.altsub.as_perm_max_lower}
  \rocqok
  $(\mathrm{turn\_count}(\sigma)).+2 \leq \mathrm{as\_perm\_max}(\sigma)$.

  Witness: $I = \{0, n+1\} \cup \{(\mathrm{val}\,t).+1 : t \text{ is a turning point of }\sigma\}$.
  Cardinality is $\mathrm{turn\_count}(\sigma) + 2$ (boundary endpoints
  plus one position per turn).  Alternation is verified between
  consecutive picks via the inter-turn monotonicity lemma applied to
  the no-turn slot interval between them.
\end{theorem}

\begin{theorem}[Upper bound]
  \label{thm:as_perm_max_upper}
  \uses{def:turn_count, def:as_perm_max}
  \rocq{mathcomp_eulerian.altsub.as_perm_max_upper}
  \rocqok
  $\mathrm{as\_perm\_max}(\sigma) \leq (\mathrm{turn\_count}(\sigma)).+2$.

  Proved via the global bound chain: \texttt{flip\_count} (number of
  direction changes in a boolean sequence) is monotone under
  subsequence, so $\mathrm{flip\_count}(\mathrm{sign\_seq}(\mathrm{pick\_seq}\,\sigma\,I))
  \leq \mathrm{flip\_count}(\mathrm{sign\_seq}(\mathrm{perm\_seq}\,\sigma))$.
  The right-hand side equals $\mathrm{turn\_count}(\sigma)$ via the
  bridge \rocq{mathcomp_eulerian.altsub.sign_seq_perm_seq}.
\end{theorem}

\begin{corollary}[Alternating-subsequence characterization]
  \label{cor:as_perm_max_eq}
  \uses{thm:as_perm_max_lower, thm:as_perm_max_upper}
  \rocq{mathcomp_eulerian.altsub.as_perm_max_eq}
  \rocqok
  \emph{Stanley EC1 \S1.6.2.}  For every $\sigma \in \Snp{n+2}$,
  \[
    \mathrm{as}(\sigma) \;=\; \mathrm{turn\_count}(\sigma) + 2.
  \]
  The longest alternating subsequence has length exactly two more
  than the number of turning points: one element from each ``monotone
  run'' of $\sigma$, plus one boundary element.

  \emph{Formalization remark.}  This took five sequential interactive
  subagent sessions.  The naive ``leftmost turn in
  $[\sigma_i, \sigma_{i+2})$'' injection (mapping each subseq sign-flip
  to a turn) is \emph{not} injective — adjacent flips can share a
  turn-window.  Session B-4 pivoted from per-interval to a global
  bound via the subseq-monotonicity of $\mathrm{flip\_count}$.
\end{corollary}

\section{André's reflection method (Stanley \S1.6.4)}
% ============================================================

\emph{Status.}  This section records a \textbf{partial formalization}.
The development lives in \code{experimental/reflection.v}, which is
\textbf{not in the active build chain} (excluded from
\code{\_CoqProject}).  The headline recurrence \texttt{euler\_rec} is
proven from one named, precisely-stated \texttt{Admitted} lemma
(\texttt{sum\_set\_is\_alt\_eq\_andre\_sum}).  The companion document
\code{docs/internal/PHASE\_C\_RETROSPECTIVE.md} captures the eight
sessions of work, the structural obstacle (a 4-to-1 boundary
reduction, not the 2-to-1 fibration suggested by the classical English
proof), and a concrete $\sim 510$ LOC discharge plan.

\begin{definition}[Euler numbers]
  \label{def:eulerA}
  Let $\mathrm{alt\_desc\_set}\,n \subseteq \{0, \dots, n-1\}$ denote
  the alternating descent pattern $\{0, 2, 4, \dots\}$.  Define
  \[
    E_{n+1}
    \;:=\;
    \beta(\mathrm{alt\_desc\_set}\,n),
    \qquad
    A_{n+1} := E_{n+1},
    \qquad
    A_0 := 1.
  \]
  $A_n$ is the number of alternating permutations of $\Snp{n}$.
\end{definition}

\begin{theorem}[The André recurrence — modulo one admit]
  \label{thm:euler_rec}
  \uses{def:eulerA}
  For every $n \in \mathbb{N}$,
  \[
    2 \cdot A_{n+2}
    \;=\;
    \sum_{k=0}^{n+1} \binom{n+1}{k} \, A_k \, A_{n+1-k}.
  \]

  \emph{Proof status.}  Derived in one line from \texttt{two\_eulerA\_split}
  (which uses \texttt{beta\_compl} from \texttt{beta\_swap.v}) and the
  named admit:
  \begin{equation}
    \label{eq:andre-admit}
    \sum_{t \in \Snp{n+1}} \sum_{p \in \{0,\dots,n+1\}}
       \mathrm{set\_is\_alt}(\mathrm{descent\_set}(\mathrm{insert\_max\_perm}\,t\,p))
    \;=\;
    \sum_{k=0}^{n+1} \binom{n+1}{k} A_k A_{n+1-k}.
  \end{equation}

  \emph{Sanity.}  The base case $n = 0$ (i.e.\ $2 A_2 = 2$) is proven
  unconditionally as \texttt{euler\_rec\_n0\_direct} (no admit used).
  The cases $n = 1, 2$ are stated via \texttt{euler\_rec} and inherit
  the admit transitively.
\end{theorem}

\begin{remark}[The structural obstacle]
  The remaining admit \eqref{eq:andre-admit} is the combinatorial heart
  of André's reflection.  After partitioning $\sigma \in \Snp{n+2}$ by
  the position $p$ of the maximum and the value-set $L \subset [n+1]$
  of size $p$ left of the maximum, the descent at the boundary slot
  $p-1$ depends on $(L, s_L, s_R)$ \emph{jointly}: specifically, on
  whether the largest left value (selected by $s_L$) exceeds the
  smallest right value (selected by $s_R$).  This couples the three
  decomposition components and prevents the $[\mathrm{descent\_set} = E]$
  indicator from factoring as $[s_L\text{-cond}] \cdot [s_R\text{-cond}]$.

  The classical proof absorbs this coupling via the factor of $2$ on
  the LHS: summing over alternating-and-anti-alternating descent
  patterns leaves the boundary parity free.  Formally, of the four
  flavour pairs $(s_L\text{-flavour}, s_R\text{-flavour}) \in
  \{\mathrm{alt}, \neg\mathrm{alt}\}^2$, exactly one is consistent
  with the forced boundary value for each fixed $(L, s_L, s_R)$ — a
  4-to-1 reduction whose formalization is estimated at $\sim 510$ LOC.
  See \code{docs/internal/PHASE\_C\_RETROSPECTIVE.md} for the full
  lemma list and discharge plan.
\end{remark}

\begin{remark}[Reusable infrastructure]
  Although the headline is open, the file
  \code{experimental/reflection.v} contains $\sim 1500$ LOC of
  axiom-free, kernel-validated infrastructure for any future
  position-of-max bijective decomposition of permutations:
  \begin{itemize}
    \item \texttt{descent\_set\_insert\_max\_*} — descent set under
      max-insertion at the three position cases.
    \item \texttt{embed\_left/right}, \texttt{perm\_left/right} —
      standardisation of left/right sub-words to $\Snp{j}$ and
      $\Snp{n+1-j}$.
    \item \texttt{descent\_set\_decomp\_partition} — the L-part /
      boundary / R-part decomposition.
    \item \texttt{assemble\_perm} + \texttt{assemble\_decomp\_inverse}
      — the full bijection $\Snp{n+1} \simeq \bigsqcup_j \{L : |L|=j\}
      \times \Snp{j} \times \Snp{n+1-j}$, both directions.
    \item \texttt{sum\_partition\_image\_left}, \texttt{sum\_reindex\_inner}
      — bigop reindexing along the bijection.
  \end{itemize}
  These can be copied into a future formalization that needs the
  position-of-max decomposition without inheriting the open admit.
\end{remark}
